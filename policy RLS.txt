-- 2. Políticas de leitura dos catálogos

-- 3. Escrita em barco somente pela equipe autenticada no papel correto

DROP POLICY IF EXISTS barco_inserir_gestao ON public.barco;
CREATE POLICY barco_inserir_gestao ON public.barco
FOR INSERT TO app_gestao_1, app_gestao_2
WITH CHECK (
    (current_user = 'app_gestao_1'
     AND app_nivel_usuario() = 'gestao_1')
    OR
    (current_user = 'app_gestao_2'
     AND app_nivel_usuario() = 'gestao_2')
);

DROP POLICY IF EXISTS barco_atualizar_gestao ON public.barco;
CREATE POLICY barco_atualizar_gestao ON public.barco
FOR UPDATE TO app_gestao_1, app_gestao_2
USING (
    (current_user = 'app_gestao_1'
     AND app_nivel_usuario() = 'gestao_1')
    OR
    (current_user = 'app_gestao_2'
     AND app_nivel_usuario() = 'gestao_2')
)
WITH CHECK (
    (current_user = 'app_gestao_1'
     AND app_nivel_usuario() = 'gestao_1')
    OR
    (current_user = 'app_gestao_2'
     AND app_nivel_usuario() = 'gestao_2')
);

-- 4. Auditoria: leitura exclusiva da gestão autenticada

DROP POLICY IF EXISTS auditoria_leitura_gestao
ON public.auditoria_sistema;

CREATE POLICY auditoria_leitura_gestao
ON public.auditoria_sistema
FOR SELECT TO app_gestao_1, app_gestao_2
USING (
    (current_user = 'app_gestao_1'
     AND app_nivel_usuario() = 'gestao_1')
    OR
    (current_user = 'app_gestao_2'
     AND app_nivel_usuario() = 'gestao_2')
);
